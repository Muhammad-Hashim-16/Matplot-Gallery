# NOTE: Please import relevant libraries (e.g., import matplotlib.pyplot as plt, import numpy as np) before running this snippet.

fig = plt.figure()
ax = fig.add_subplot(111, projection='3d')

t = np.linspace(0, 20, 200)
x = np.sin(t)
y = np.cos(t)
z = t

ax.plot(x, y, z, color="red")
ax.set_title("3D Line Plot")
plt.show()
