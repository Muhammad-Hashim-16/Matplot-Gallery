# NOTE: Please import relevant libraries (e.g., import matplotlib.pyplot as plt, import numpy as np) before running this snippet.

fig = plt.figure(figsize=(10, 8))

for i in range(1, 10):
    plt.subplot(3, 3, i)
    plt.plot(np.random.randn(10))
    plt.title(f"Plot {i}")

plt.tight_layout()
