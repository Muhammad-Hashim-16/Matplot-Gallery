# NOTE: Please import relevant libraries (e.g., import matplotlib.pyplot as plt, import numpy as np) before running this snippet.

x = np.linspace(0, 10, 200)
y = np.sin(x) * np.exp(-x/5)

plt.figure(figsize=(9,5), dpi=120)
plt.plot(x, y, color="black", linewidth=2, label="Damped sine")
plt.fill_between(x, y, color="gray", alpha=0.3)
plt.title("Area Under Curve Concept")
plt.legend()
plt.grid(True)
plt.show()
